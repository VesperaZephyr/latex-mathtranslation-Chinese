% 注意: after-code 在 group 内执行, 须用 \gdef 使重定义全局生效
\newcommand{\AJseccount}{%

}

  \setcounter{theorem}{0}%
  \gdef\thetheorem{\thesubsection.\arabic{theorem}}%

}
